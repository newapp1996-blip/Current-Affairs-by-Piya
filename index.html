let appData = typeof appData !== 'undefined' ? appData : { news: [], quizzes: [] }; async function loadData() { // If appData is empty, fetch data.json if (!appData.news || appData.news.length === 0) { try { const res = await fetch('data.json?v=' + new Date().getTime()); if (res.ok) { appData = await res.json(); } } catch (e) { console.error("Error loading data.json", e); } } renderNews(); } function renderNews() { const container = document.getElementById('news-container') || document.querySelector('.news-feed'); if (!container) return; if (!appData.news || appData.news.length === 0) { container.innerHTML = `
No news available. Run the GitHub Action workflow to fetch updates.
`; return; } container.innerHTML = appData.news.map(item => `
[${item.category}]

${item.title}

📍 ${item.place} | 🗓️ ${item.date}

${item.conclusion}

`).join(''); } document.addEventListener('DOMContentLoaded', loadData);
